const { PDFDocument } = require('pdf-lib');
const fs = require('fs');
const path = require('path');

async function stitchImagesToPDF(imageFolder, outputFile) {
  console.log(`Starting to stitch images from ${imageFolder}...`);
  const pdfDoc = await PDFDocument.create();
  
  // Read and sort files chronologically (assumes format like page_001.png, page_002.png)
  const files = fs.readdirSync(imageFolder)
    .filter(file => file.endsWith('.png'))
    .sort(); 

  for (const file of files) {
    const imageBytes = fs.readFileSync(path.join(imageFolder, file));
    const image = await pdfDoc.embedPng(imageBytes);
    
    // Create a page matching the exact dimensions of the screenshot
    const page = pdfDoc.addPage([image.width, image.height]);
    page.drawImage(image, {
      x: 0,
      y: 0,
      width: image.width,
      height: image.height,
    });
    console.log(`Embedded ${file}`);
  }

  const pdfBytes = await pdfDoc.save();
  fs.writeFileSync(outputFile, pdfBytes);
  console.log(`✅ Success! Stitched ${files.length} pages into ${outputFile}`);
}

// Execute the stitcher (Update paths as needed)
const inputFolder = path.join(__dirname, 'screenshots');

Heard, Chef! You just got caught by two classic Zsh terminal traps at the exact same time. It has nothing to do with the code Gemini just refactored; it's purely how your Mac/Linux terminal interprets special characters.
Here is exactly what went wrong and how to fix it:
The URL Trap (zsh: no matches found): Zsh uses the ? and & characters in your Google Books URL for pattern matching and background processing. Because you didn't put the URL in quotes, Zsh tried to execute it as a terminal command instead of passing it to your CLI script.
The Flag Trap (zsh: =pages=951 not found): You accidentally typed a double-equals sign (==pages) instead of a double-dash (--pages). In Zsh, an equals sign triggers absolute path expansion, which immediately crashed.
Here is your exact, corrected command. Notice the quotes around the URL and the dashes on the pages flag:
pnpm start import:textbook "https://play.google.com/books/reader?id=E32Rk9Hp-AYC&pg=GBS.PA1" --slug=flavor-bible --pages=951
